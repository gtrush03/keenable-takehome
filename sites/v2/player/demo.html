Mount
<link rel="stylesheet" href="player/player.css">
<script src="player/player.js"></script>

KPlayer.mount(el, {
  src: "media/fintech.mp4",
  poster: "media/fintech.jpg",
  title: "Search with a clock",
  chapters: "fintech",   // or [{t, label}]
  autoplayMuted: false, loop: false,
  fill: false            // true = fill a sized slide
});

KPlayer.hero(el, {
  loopSrc: "media/hero-loop.mp4",
  poster: "media/fintech.jpg",
  eyebrow: "A founding GTM take-home",
  title: "Search with a clock.",
  films: [{ label: "Watch the film",
    src: "media/fintech.mp4",
    poster: "media/fintech.jpg",
    title: "Search with a clock",
    chapters: "fintech" }]
});

KPlayer.open({ src, poster, title, chapters });
Keys

space / k: play, pause
j / l: back, forward 10 s
shift + > / <: speed 1× · 1.5× · 2× · hold the picture: 2×
f: full screen · m: mute · c: chapters
← / →: never taken by the player (slides)
Esc: close the chapter list or the overlay
